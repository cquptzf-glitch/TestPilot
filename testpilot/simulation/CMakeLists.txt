cmake_minimum_required(VERSION 3.21)
project(TestPilotMockInstrument VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Quick Qml)
qt_standard_project_setup()
qt_add_executable(MockInstrument WIN32 MockInstrument.cpp QtFeatures.cpp QtFeatures.h)
qt_add_resources(MockInstrument "qt_lab" PREFIX "/qt-lab" FILES QtQuickLab.qml)
target_link_libraries(MockInstrument PRIVATE Qt6::Widgets Qt6::Network Qt6::Quick Qt6::Qml)
if(MSVC)
  target_compile_options(MockInstrument PRIVATE /utf-8)
endif()
